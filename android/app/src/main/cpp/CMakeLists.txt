cmake_minimum_required(VERSION 3.22)
project(mirage_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
FetchContent_Declare(llama
  URL https://github.com/ggml-org/llama.cpp/archive/931351ea50dfdd3ee249606f655eef2e9a629daf.tar.gz)
FetchContent_MakeAvailable(llama)
if(ANDROID)
  add_library(mirage_voice SHARED mirage_voice.cpp)
  target_link_libraries(mirage_voice PRIVATE llama android log)
  target_link_options(mirage_voice PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(mirage_voice mirage_voice.cpp)
  target_compile_definitions(mirage_voice PRIVATE MIRAGE_CLI=1)
  target_link_libraries(mirage_voice PRIVATE llama)
endif()
